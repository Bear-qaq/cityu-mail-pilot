# The landing page is meant to be found; the application is not.
#
# `/` is prose a stranger should be able to search for, and it is the only page
# worth indexing. `/app` is a login shell: a crawler that renders it gets an
# empty application frame, and one that indexes it puts a page in the results
# that a visitor cannot do anything with. `/api/` and `/download/` are not
# pages at all.
#
# There is deliberately no `Sitemap:` line: this site is one indexable URL, and
# a sitemap listing a single page is noise that has to be kept in sync for no
# benefit. Add one if a second public page ever appears.
User-agent: *
Allow: /
Disallow: /app
Disallow: /app.js
Disallow: /api/
Disallow: /download/
Disallow: /.well-known/
